API 키를 확인하거나 다시 시도해 주세요."); nextStep(3); } } function displayResults(data) { const userFault = data.userFaultRatio || 50; const friendFault = data.friendFaultRatio || 50; document.getElementById('userFaultBar').style.width = userFault + '%'; document.getElementById('userFaultBar').innerText = `나 ${userFault}%`; document.getElementById('friendFaultBar').style.width = friendFault + '%'; document.getElementById('friendFaultBar').innerText = `친구 ${friendFault}%`; document.getElementById('faultReason').innerText = data.faultReason; document.getElementById('toneAdvice').innerText = data.recommendedTone; document.getElementById('placeAdvice').innerText = data.recommendedPlaceTime; const scriptListContainer = document.getElementById('scriptList'); scriptListContainer.innerHTML = ''; data.scripts.forEach((script, index) => { const card = document.createElement('div'); card.className = `script-card ${index === 0 ?
